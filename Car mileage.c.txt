
#include <stdio.h>

int main() {
    float mileage;
    int petrol, distance;
    float availableDistance;

    scanf("%f", &mileage);
    scanf("%d", &petrol);
    scanf("%d", &distance);

    availableDistance = mileage * petrol;

    if (availableDistance >= distance)
        printf("Can reach");
    else
        printf("Cannot reach");

    return 0;
}
