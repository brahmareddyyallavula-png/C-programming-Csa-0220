
#include <stdio.h>

int main() {
    int coins, x, y;
    int longBen, blackbeard, remaining, each;

    scanf("%d", &coins);
    scanf("%d", &x);
    scanf("%d", &y);

    longBen = coins * x / 100;
    remaining = coins - longBen;

    blackbeard = remaining * y / 100;
    remaining -= blackbeard;

    each = remaining / 4;

    printf("%d\n", longBen);
    printf("%d\n", blackbeard);
    printf("%d\n", each);

    return 0;
}
