
#include <stdio.h>

int main() {
    int A, B, C, cars;

    scanf("%d%d%d%d", &A, &B, &C, &cars);

    if (cars != 0 && A % cars == 0)
        printf("Car 1 goes into road A");
    else if (cars != 0 && B % cars == 0)
        printf("Car 1 goes into road B");
    else if (cars != 0 && C % cars == 0)
        printf("Car 1 goes into road C");
    else
        printf("No path exists");

    return 0;
}
